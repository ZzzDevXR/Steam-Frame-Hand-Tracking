cmake_minimum_required(VERSION 3.22)
project(framehand_mercury VERSION 25.1.0 LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 20)
set(MONADO_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/mercury/monado" CACHE PATH "Pinned, patched Monado tree (mercury/build.sh)")
include(${MONADO_SOURCE_DIR}/cmake/MonadoSetOptions.cmake)
list(APPEND CMAKE_MODULE_PATH "${MONADO_SOURCE_DIR}/cmake")
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(ONNXRuntime REQUIRED MODULE)
find_package(OpenCV REQUIRED CONFIG)
monado_set_options(ENABLE XRT_HAVE_OPENCV XRT_HAVE_ONNXRUNTIME XRT_MODULE_MERCURY_HANDTRACKING
                  KEEP XRT_HAVE_SYSTEM_CJSON XRT_HAVE_LIBUDEV XRT_HAVE_LINUX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(${MONADO_SOURCE_DIR} monado EXCLUDE_FROM_ALL)
set_property(TARGET aux_tracking PROPERTY SOURCES
    "${MONADO_SOURCE_DIR}/src/xrt/auxiliary/tracking/t_camera_models.cpp"
    "${MONADO_SOURCE_DIR}/src/xrt/auxiliary/tracking/t_data_utils.c")
foreach(_target aux_tracking t_ht_mercury t_ht_mercury_model t_ht_mercury_distorter t_ht_mercury_kine_lm)
  foreach(_property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
    get_target_property(_links ${_target} ${_property})
    if(_links)
      list(FILTER _links EXCLUDE REGEX "opencv_(highgui|imgcodecs|video|calib3d|features2d|flann)")
      set_property(TARGET ${_target} PROPERTY ${_property} "${_links}")
    endif()
  endforeach()
endforeach()
# The patched hg_model.cpp includes fhm_nn.h (monado-executorch.patch). Without FHM_NN_CMAKE_DIR the ExecuTorch
# path is compiled out and FHL_NN_BACKEND=executorch reports an error. FHM_NN_CMAKE_DIR names a CMake project
# that defines the static library target fhm_nn (fhm_nn.cpp with FHM_EXECUTORCH=1 plus the ExecuTorch runtime).
foreach(_target t_ht_mercury t_ht_mercury_model t_ht_mercury_distorter t_ht_mercury_kine_lm)
  target_include_directories(${_target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
endforeach()
set(FHM_NN_CMAKE_DIR "" CACHE PATH "Optional CMake project providing an ExecuTorch-enabled fhm_nn target")
if(FHM_NN_CMAKE_DIR)
  add_subdirectory("${FHM_NN_CMAKE_DIR}" "${CMAKE_BINARY_DIR}/fhm_nn")
  add_library(framehand_mercury SHARED framehand_mercury.cpp)
else()
  add_library(framehand_mercury SHARED framehand_mercury.cpp fhm_nn.cpp)
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/diagnostics.patch" _trace_hash)
target_compile_definitions(framehand_mercury PRIVATE FHM_DIAGNOSTICS_PATCH_SHA256="${_trace_hash}")
option(FHM_EXPERIMENTAL_GATE "Explicitly enable the separately labeled gate-only research fork" OFF)
if(FHM_EXPERIMENTAL_GATE)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/experimental-acquisition-gate.patch" _gate_hash)
  target_compile_definitions(framehand_mercury PRIVATE FHM_EXPERIMENTAL_GATE=1 FHM_ACCEPTANCE_PATCH_SHA256="${_gate_hash}")
endif()
target_include_directories(framehand_mercury PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(framehand_mercury PRIVATE t_ht_mercury ONNXRuntime::ONNXRuntime Eigen3::Eigen)
if(FHM_NN_CMAKE_DIR)
  target_link_libraries(framehand_mercury PRIVATE t_ht_mercury_model fhm_nn)
endif()
if(UNIX AND NOT APPLE)
  target_link_options(framehand_mercury PRIVATE "-Wl,-z,defs"
    "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/mercury.exports")
endif()
set_target_properties(framehand_mercury PROPERTIES BUILD_RPATH_USE_ORIGIN ON)
set(FHM_LIBRARY_BASENAME "framehand_mercury" CACHE STRING "Unique artifact basename")
set_target_properties(framehand_mercury PROPERTIES OUTPUT_NAME "${FHM_LIBRARY_BASENAME}")
